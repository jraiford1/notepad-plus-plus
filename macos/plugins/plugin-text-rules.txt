# Text rules for plugins: the places where code many plugins share (Notepad++'s plugin template and
# its copies) relies on what Visual C++ allows and clang does not, once wide strings are char16_t.
# npp-migrate applies them to every plugin (--plugin-text-rules): '*' matches across folders, and a
# rule that changes nothing in a plugin is no error. The format is text-rules.txt's.

# The template's setCommand takes the command's name as a TCHAR *, and is called with literals: Visual
# C++ converts a wide literal to a pointer to non-const, clang does not for char16_t. It only copies
# the name, so it takes a pointer to const.
files: *PluginDefinition.*
line: setCommand(
regex: ([(,]\s*)(TCHAR|WCHAR|wchar_t)\s*\*\s*cmdName => $1const $2 *cmdName

# A macro that makes a wide string of its argument (L#x, after Visual C++'s _CRT_WIDE) pastes the L
# prefix onto the stringized argument; the u"" before it does the same for char16_t, by concatenation.
files: *
line: L#
regex: \bL#\s*(\w+) => u"" #$1

# A literal given to a structure's text pointer (LVCOLUMN, TVITEM... pszText), which is not const in
# Windows' headers: the controls only read it there.
files: *
line: pszText
regex: (pszText\s*=\s*)((?:TEXT|_T)\s*\(\s*"(?:[^"\\]|\\.)*"\s*\)|L"(?:[^"\\]|\\.)*") => $1const_cast<LPTSTR>($2)

# The same for a menu item's text (MENUITEMINFO dwTypeData), which InsertMenuItem only reads.
files: *
line: dwTypeData
regex: (dwTypeData\s*=\s*)((?:TEXT|_T)\s*\(\s*"(?:[^"\\]|\\.)*"\s*\)|L"(?:[^"\\]|\\.)*") => $1const_cast<LPTSTR>($2)
