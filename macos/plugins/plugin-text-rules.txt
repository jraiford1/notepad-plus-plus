# Text rules for plugins: the places where code many plugins share (Notepad++'s plugin template and
# its copies) relies on what Visual C++ allows and clang does not, once wide strings are char16_t.
# npp-migrate applies them to every plugin (--plugin-text-rules): '*' matches across folders, and a
# rule that changes nothing in a plugin is no error. The format is text-rules.txt's.

# The template's setCommand takes the command's name as a TCHAR *, and is called with literals: Visual
# C++ converts a wide literal to a pointer to non-const, clang does not for char16_t. It only copies
# the name, so it takes a pointer to const.
files: *PluginDefinition.*
line: setCommand(
regex: ([(,]\s*)(TCHAR|WCHAR|wchar_t)\s*\*\s*cmdName => $1const $2 *cmdName

# A macro that makes a wide string of its argument (L#x, after Visual C++'s _CRT_WIDE) pastes the L
# prefix onto the stringized argument; the u"" before it does the same for char16_t, by concatenation.
files: *
line: L#
regex: \bL#\s*(\w+) => u"" #$1

# A literal given to a structure's text pointer (LVCOLUMN, TVITEM... pszText), which is not const in
# Windows' headers: the controls only read it there.
files: *
line: pszText
regex: (pszText\s*=\s*)((?:TEXT|_T)\s*\(\s*"(?:[^"\\]|\\.)*"\s*\)|L"(?:[^"\\]|\\.)*") => $1const_cast<LPTSTR>($2)

# NppExport's exporters return their clipboard format's name, a literal, as TCHAR *.

# XBrackets stringizes its version, "2, 0, 4, 0": a macro argument with commas; it has the string too.

replace: STR(PLUGIN_VERSION) => PLUGIN_VERSION_STR

# GitSCM shows Windows Explorer's context menu for the files of its panel, through the shell namespace,

files: GitSCM/DockingFeature/GitPanelDlg.cpp

# DoxyIt's stringReplace changes a string in place, and is given temporaries (Visual C++ binds them to
# a reference to non-const): for those, an overload that takes the string and returns it.
files: DoxyIt/src/Utilities/Utils.h
line: std::string& stringReplace(
replace: const std::string& newStr); => const std::string& newStr); inline std::string stringReplace(std::string&& str, const std::string& oldStr, const std::string& newStr) { stringReplace(str, oldStr, newStr); return std::move(str); }

# ColumnTools reads a setting into &static_cast<int>(g_iEdgeColOrig): Visual C++ gives the address of a
# temporary copy, so the value read is dropped. The same here: a temporary that lasts the call.
files: ColumnTools/HorizontalRuler.cpp
line: &static_cast<int>(g_iEdgeColOrig)
replace: &static_cast<int>(g_iEdgeColOrig) => [](int&& copy) { return &copy; }(static_cast<int>(g_iEdgeColOrig))

# NppSnippets and NppTags open their SQLite databases by Windows paths ('\' separators), which the
# file functions take but SQLite's own open() does not: they open them by the macOS path.
files: NppSnippets/SqliteDB.cpp
line: sqlite3_open16(_dbFile.c_str(), &_db)
replace: sqlite3_open16(_dbFile.c_str(), &_db) => sqlite3_open(nppmac::nativePath(std::u16string_view{_dbFile.c_str()}).c_str(), &_db)

files: NppTags/SqliteDB.cpp
line: sqlite3_open16(_dbFile, &_db)
replace: sqlite3_open16(_dbFile, &_db) => sqlite3_open(nppmac::nativePath(std::u16string_view{_dbFile}).c_str(), &_db)
