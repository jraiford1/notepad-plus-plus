# The Win32 runtime: macOS implementations of the Windows API subset Notepad++ uses.
#
# Declarations come from Wine's headers (see wine/WineHeaders.cmake); include/ comes first so the
# runtime can supply or replace a header where Wine's does not fit macOS.

add_library(nppRuntimeHeaders INTERFACE)
target_include_directories(nppRuntimeHeaders INTERFACE
	${CMAKE_CURRENT_SOURCE_DIR}/include
	${NPPMAC_WINE_INCLUDE_DIRS}
)
target_compile_definitions(nppRuntimeHeaders INTERFACE
	UNICODE _UNICODE NOMINMAX OEMRESOURCE _USE_64BIT_TIME_T
	_WIN32_WINNT=0x0601 NTDDI_VERSION=0x06010000
	NPPMAC=1
)
target_compile_options(nppRuntimeHeaders INTERFACE
	"SHELL:-include \"${CMAKE_CURRENT_SOURCE_DIR}/include/nppmac/ForcedInclude.h\""
)

# --- COM interface identifiers (Wine's uuid library, generated by widl) ------------------------
file(GLOB nppUuidSources "${NPPMAC_WINE_UUID_DIR}/*_i.c")
add_library(nppUuid STATIC ${nppUuidSources})
target_link_libraries(nppUuid PRIVATE nppRuntimeHeaders)
target_compile_options(nppUuid PRIVATE -w)

file(GLOB_RECURSE nppRuntimeSources CONFIGURE_DEPENDS
	"${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp"
	"${CMAKE_CURRENT_SOURCE_DIR}/src/*.mm"
	"${CMAKE_CURRENT_SOURCE_DIR}/src/*.h"
)
# Code for one platform lives in folders named after it (src/platform/macos...): only this platform's.
list(FILTER nppRuntimeSources EXCLUDE REGEX "/(macos|linux)/")
file(GLOB_RECURSE nppRuntimePlatformSources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*/${NPPMAC_PLATFORM}/*")
if(NPPMAC_PLATFORM STREQUAL "macos")
	list(APPEND nppRuntimeSources ${nppRuntimePlatformSources})
else()
	# Linux's native side (Qt, POSIX) is compiled without the Win32 headers and the forced include, as
	# AppKit's .mm side is kept apart from them on the Mac.
	add_library(nppRuntimeNative OBJECT ${nppRuntimePlatformSources})
	target_include_directories(nppRuntimeNative PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
	target_link_libraries(nppRuntimeNative PUBLIC Qt6::Core Qt6::Gui Qt6::Widgets Qt6::DBus)
endif()
if(NPPMAC_PLATFORM STREQUAL "linux")
	# Not on Linux until J4's editor step gives it Scintilla's Qt layer: the Scintilla window class.
	list(FILTER nppRuntimeSources EXCLUDE REGEX "/scintilla/ScintillaClass\\.cpp$")
endif()
add_library(nppRuntime STATIC ${nppRuntimeSources})
if(NPPMAC_PLATFORM STREQUAL "linux")
	target_sources(nppRuntime PRIVATE $<TARGET_OBJECTS:nppRuntimeNative>)
endif()
target_include_directories(nppRuntime PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
# AppKit code (.mm) and Win32 code (.cpp) never share a translation unit: Cocoa's headers and the
# Windows headers both define BOOL, Polygon, GetCurrentProcess and more.
set_source_files_properties(${nppRuntimeSources} PROPERTIES COMPILE_OPTIONS "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
# The COM interface identifiers (nppUuid, made with the app) for the runtime's own COM objects.
target_link_libraries(nppRuntime PUBLIC nppUuid)
if(NPPMAC_PLATFORM STREQUAL "macos")
	target_link_libraries(nppRuntime PUBLIC nppRuntimeHeaders "-framework Cocoa" "-framework Carbon" "-framework QuartzCore" "-framework IOSurface" "-framework CoreText" "-framework PDFKit" "-framework Metal" iconv)
	# The Scintilla window class talks to Scintilla's Cocoa view.
	target_include_directories(nppRuntime PRIVATE "${NPP_ROOT}/scintilla/include" "${NPP_ROOT}/scintilla/cocoa")
else()
	target_link_libraries(nppRuntime PUBLIC nppRuntimeHeaders Qt6::Core Qt6::Gui Qt6::Widgets Qt6::DBus ${CMAKE_DL_LIBS})
	target_include_directories(nppRuntime PRIVATE "${NPP_ROOT}/scintilla/include")
endif()
